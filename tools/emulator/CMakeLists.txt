cmake_minimum_required(VERSION 3.16)
project(emerald_mgba_bridge C)
set(CMAKE_C_STANDARD 11)
find_path(MGBA_INCLUDE_DIR mgba/core/core.h REQUIRED)
find_library(MGBA_LIBRARY NAMES mgba REQUIRED)
add_executable(emerald-mgba-bridge bridge.c)
target_include_directories(emerald-mgba-bridge PRIVATE ${MGBA_INCLUDE_DIR})
target_link_libraries(emerald-mgba-bridge PRIVATE ${MGBA_LIBRARY})
target_compile_options(emerald-mgba-bridge PRIVATE -Wall -Wextra -Werror)
